struct node{
		struct node *lc;
		char w;
		struct node *rc;
		};

struct node* create(char exp[40])
{
		struct node *stack[strlen(exp)];
		int top=-1;
		for(int i=0;i<strlen(exp);i++)
		{
			struct node* new=malloc(sizeof(struct node *));
			new->w=exp[i];
			new->lc=NULL;
			new->rc=NULL;
			if(exp[i]=='+'|| exp[i]=='-' || exp[i]=='*' || exp[i]=='/' || exp[i]=='^')
			{

				new->rc=stack[top--];
				new->lc=stack[top--];
				stack[++top]=new;
			}
			else
			{
				top++;
				stack[top]=new;
			}
		}
		return stack[top];
}

#include<stdio.h>

struct node{
	struct node *lc;
	char w[40];
	char m[40];
	struct node *rc;
	};

struct node* insert(struct node *root,char word[40],char meaning[40])
{
	if(root==NULL)
	{
			struct node* new=(struct node *)malloc(1*sizeof(struct node *));
			new->lc=NULL;
			new->rc=NULL;
			strcpy(new->w,word);
			strcpy(new->m,meaning);
			root=new;
	}
	else
	{
		int flag=0;
		struct node* ptr=root;
		struct node*ptr1;
		while((ptr!=NULL)&&(flag!=1))
		{
			if(strcmp(ptr->w,word)>0)
			{
				ptr1=ptr;
				ptr=ptr->lc;
			}
			else if(strcmp(ptr->w,word)<0)
			{
				ptr1=ptr;
				ptr=ptr->rc;
			}
			else
			{
				flag=1;
				printf("the word is already present");
			}
		}
		if(flag==0)
		{
			struct node* new=(struct node *)malloc(1*sizeof(struct node *));
			new->lc=NULL;
			new->rc=NULL;
			strcpy(new->w,word);
			strcpy(new->m,meaning);

			if(strcmp(ptr1->w,word)>0)
			{
				ptr1->lc=new;
			}
			else if(strcmp(ptr1->w,word)<0)
			{
				ptr1->rc=new;
			}
		}
	}
	return root;
}

}
